local preloadedData = {
    profile = nil,
    repos = nil,
    repoContents = {},

local preloadStarted = false
local repoPreloadStarted = false
local repoQueue = {}
local isRepoPreloading = false

function startInitialPreload(token, userName)
    if preloadStarted then return end
    if not token or token == "" or not userName or userName == "" then return end
    preloadStarted = true
    
    local headers = {
        ["Authorization"] = "Bearer " .. token,
        ["Accept"] = "application/vnd.github.v3+json",
        ["User-Agent"] = "GitHub-Manager-App"
    }
    
    Http.get("https://api.github.com/user", headers, function(code, body)
        if code == 200 then preloadedData.profile = body end
        
        Handler(Looper.getMainLooper()).postDelayed(Runnable({
            run = function()
                Http.get("https://api.github.com/user/repos?per_page=100", headers, function(rCode, rBody)
                    if rCode == 200 then preloadedData.repos = rBody end
                end)
            end
        }), 200)
    end)
end

function processNextRepoContent(token, userName)
    if #repoQueue == 0 then
        isRepoPreloading = false
        return
    end
    isRepoPreloading = true
    local rName = repoQueue[1]
    table.remove(repoQueue, 1)
    
    local headers = {
        ["Authorization"] = "Bearer " .. token,
        ["Accept"] = "application/vnd.github.v3+json",
        ["User-Agent"] = "GitHub-Manager-App"
    }
    local safePath = Uri.encode("", "/")
    local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(rName) .. "/contents/" .. safePath
    
    Http.get(url, headers, function(cCode, cBody)
        if cCode == 200 then
            preloadedData.repoContents[rName .. "|"] = cBody
            local sC, files = pcall(function() return json.decode(cBody) end)
            if sC and type(files) == "table" then
                for f = 1, #files do
                    local file = files[f]
                    if file.type == "file" and file.download_url then
                        Http.get(file.download_url, headers, function(fCode, fBody)
                            if fCode == 200 then
                                preloadedData.fileContents[file.sha] = fBody
                            end
                        end)
                    end
                end
            end
        end
        Handler(Looper.getMainLooper()).postDelayed(Runnable({
            run = function()
                processNextRepoContent(token, userName)
            end
        }), 200)
    end)
end

function startRepoContentsPreload(token, userName, reposBody)
    if repoPreloadStarted then return end
    repoPreloadStarted = true
    
    local success, repos = pcall(function() return json.decode(reposBody) end)
    if not success or type(repos) ~= "table" then return end
    
    for i = 1, #repos do
        if repos[i].name then
            table.insert(repoQueue, repos[i].name)
        end
    end
    
    if not isRepoPreloading then
        Handler(Looper.getMainLooper()).postDelayed(Runnable({
            run = function()
                processNextRepoContent(token, userName)
            end
        }), 200)
    end
end

local function trim(s)
  if s == nil then return "" end
  local res = tostring(s):gsub("^%s*(.-)%s*$", "%1")
  res = res:gsub("[%z\1-\31]", "")
  return res
end

local function getErrorMessage(code, body)
  local msg = "Error code: " .. tostring(code)
  if body and body ~= "" then
    local success, data = pcall(function() return json.decode(body) end)
    if success and data then
      if data.message then
        msg = msg .. "\n" .. data.message
      end
      if data.errors and type(data.errors) == "table" then
        for i, err in ipairs(data.errors) do
          msg = msg .. "\n- " .. (err.message or "")
        end
      end
    else
      msg = msg .. "\n" .. body
    end
  end
  return msg
end

local function customHttpRequest(urlStr, method, reqHeaders, reqBody, callback)
  local headersJson = json.encode(reqHeaders)
  task(function(urlStr, method, headersJson, reqBody)
    require "import"
    import "java.net.URL"
    import "java.io.InputStreamReader"
    import "java.io.BufferedReader"
    import "java.lang.StringBuilder"
    import "java.lang.String"
    
    local cjson = nil
    if pcall(require, "cjson") then cjson = require("cjson") else cjson = require("json") end
    
    local code = 0
    local respStr = ""
    local success, err = pcall(function()
      local url = URL(urlStr)
      local conn = url.openConnection()
      
      if method == "PATCH" then
        conn.setRequestMethod("POST")
        conn.setRequestProperty("X-HTTP-Method-Override", "PATCH")
      else
        conn.setRequestMethod(method)
      end
      
      local decodedHeaders = cjson.decode(headersJson)
      for k, v in pairs(decodedHeaders) do
        conn.setRequestProperty(k, v)
      end
      
      if reqBody and reqBody ~= "" then
        conn.setDoOutput(true)
        local os = conn.getOutputStream()
        os.write(String(reqBody).getBytes("UTF-8"))
        os.flush()
        os.close()
      end
      
      code = conn.getResponseCode()
      local stream
      if code >= 200 and code < 300 then
        stream = conn.getInputStream()
      else
        stream = conn.getErrorStream()
      end
      if stream then
        local reader = BufferedReader(InputStreamReader(stream, "UTF-8"))
        local sb = StringBuilder()
        local line = reader.readLine()
        while line do
          sb.append(line).append("\n")
          line = reader.readLine()
        end
        reader.close()
        respStr = sb.toString()
      end
    end)
    if not success or code == 0 then
        return -1, '{"message":"No internet connection. Please check your internet connection and try again later."}'
    end
    return code, respStr
  end, urlStr, method, headersJson, reqBody, function(code, respStr)
    if callback then callback(code, respStr) end
  end)
end

function checkAndShowNewFeatures()
    local lastShownVersion = prefs.getString("lastShownVersion", "")
    if lastShownVersion ~= CURRENT_VERSION then
        customHttpRequest(getFreshUrl(NEW_FEATURES_URL), "GET", {}, "", function(code, response)

function performUpdate(mainCode, onlineVersion)
    if not mainCode or trim(mainCode) == "" then
        showUpdateErrorDialog("Update Failed", "Main code is empty.")
        return
    end
    updateInProgress = true
    
    local success = false
    local errInfo = ""
    
    local successCall, err = pcall(function()
        local f, e = io.open(TOOL_PATH, "w")
        if f then
            f:write(mainCode)
            f:close()
            success = true
        else
            errInfo = tostring(e)
        end
    end)
    
    if not success then
        pcall(function()
            require "import"
            import "java.io.File"
            import "java.io.FileOutputStream"
            import "java.lang.String"
            
            local targetFile = File(TOOL_PATH)
            local parentDir = targetFile.getParentFile()
            if not parentDir.exists() then
                parentDir.mkdirs()
            end
            
            local fos = FileOutputStream(targetFile)
            fos.write(String(mainCode).getBytes("UTF-8"))
            fos.flush()
            fos.close()
            success = true
        end)
    end
    
    if success then
        updateInProgress = false
        Handler(Looper.getMainLooper()).post(Runnable{
            run=function()
                local successDialog = AlertDialog.Builder(ctx)
                successDialog.setTitle("Update Successful")
                successDialog.setMessage("Tool successfully updated to version " .. onlineVersion .. ".")
                successDialog.setPositiveButton("Please restart your plugin manually", function()
                    if repoDialog then repoDialog.dismiss() end
                    if loginWindow then loginWindow.dismiss() end
                    prefs.edit().putString("lastShownVersion", "").apply()
                end)
                successDialog.setCancelable(false)
                successDialog.show()
            end
        })
    else
        updateInProgress = false
        showUpdateErrorDialog("Update Failed", "Failed to overwrite the update file.\n" .. errInfo)
    end
end

function checkUpdate()
    if updateInProgress then return end
    
    customHttpRequest(getFreshUrl(VERSION_URL), "GET", {}, "", function(code, response)
        if code == 200 and response then
            local onlineVersion = trim(response)
            if onlineVersion ~= CURRENT_VERSION then
                customHttpRequest(getFreshUrl(UPDATE_CODE_URL), "GET", {}, "", function(code2, mainCode)

                                updateAlertDlg.setMessage("A new version (" .. onlineVersion .. ") is available. Would you like to update now?")

                                end)

                                updateAlertDlg.show()
                            end
                        })
                    end
                end)
            else
                checkAndShowNewFeatures()
            end
        end
    end)
end

          if currentTypeFilter == "Archived" then matchType = r.archived == true

          elseif currentTypeFilter == "Private" then matchType = r.private == true
          elseif currentTypeFilter == "Public" then matchType = r.private == false
          elseif currentTypeFilter == "Source" then matchType = r.fork == false
          elseif currentTypeFilter == "Mirror" then matchType = (r.mirror_url ~= nil and r.mirror_url ~= json.null)
          elseif currentTypeFilter == "Template" then matchType = r.is_template == true
          end
          
          if matchSearch and matchType then
            table.insert(filtered, r)
          end
        end
      end
      
      table.sort(filtered, function(a, b)
        if currentSortFilter == "Recently pushed" then
          return (a.pushed_at or "") > (b.pushed_at or "")
        elseif currentSortFilter == "Least recently pushed" then
          return (a.pushed_at or "") < (b.pushed_at or "")
        elseif currentSortFilter == "Newest" then
          return (a.created_at or "") > (b.created_at or "")
        elseif currentSortFilter == "Oldest" then
          return (a.created_at or "") < (b.created_at or "")
        elseif currentSortFilter == "Name ascending" then
          return string.lower(a.name or "") < string.lower(b.name or "")
        elseif currentSortFilter == "Name descending" then
          return string.lower(a.name or "") > string.lower(b.name or "")
        elseif currentSortFilter == "Most starred" then
          return (tonumber(a.stargazers_count) or 0) > (tonumber(b.stargazers_count) or 0)
        elseif currentSortFilter == "Least starred" then
          return (tonumber(a.stargazers_count) or 0) < (tonumber(b.stargazers_count) or 0)
        end
        return false
      end)
      
      local repoNames = {}
      local repoData = {}
      repoObjects = {}
      
      for i = 1, #filtered do
        local r = filtered[i]
        local rName = tostring(r.name)
        local isPriv = r.private and "Private" or "Public"
        local lang = (r.language and r.language ~= json.null) and tostring(r.language) or "No Language"
        local updated = r.pushed_at and string.sub(r.pushed_at, 1, 10) or "Unknown"
        
        table.insert(repoNames, rName)
        table.insert(repoData, { 
          repoNameTxt = rName,
          repoVisTxt = isPriv,
          repoDetailTxt = lang .. "  |  Updated: " .. updated
        })
        repoObjects[rName] = r
      end
      
      itemsIds.txtRepoCount.setText("My All Repositories (" .. tostring(#repoNames) .. ")")
      itemsIds.btnTypeFilter.setText("Type: " .. currentTypeFilter)
      itemsIds.btnSortFilter.setText("Sort: " .. currentSortFilter)
      
      if #repoNames == 0 then
        itemsIds.repoList.setVisibility(View.GONE)
        itemsIds.emptyMsgRepo.setVisibility(View.VISIBLE)
      else
        itemsIds.repoList.setVisibility(View.VISIBLE)
        itemsIds.emptyMsgRepo.setVisibility(View.GONE)
        local adapterLayout = {
          LinearLayout, orientation = "vertical", layout_width = "fill", padding = "15dp",
          {
            LinearLayout, orientation = "horizontal", layout_width = "fill",
            { TextView, id = "repoNameTxt", textSize = "18sp", textColor = 0xFF000000, typeface = Typeface.DEFAULT_BOLD, layout_weight = "1" },
            { TextView, id = "repoVisTxt", textSize = "12sp", textColor = 0xFF666666, padding = "3dp" }
          },
          { TextView, id = "repoDetailTxt", textSize = "14sp", textColor = 0xFF555555, layout_marginTop = "5dp" }
        }
        
        adapter = LuaAdapter(ctx, repoData, adapterLayout)
        itemsIds.repoList.setAdapter(adapter)
        
        itemsIds.repoList.onItemClick = function(parent, view, position, id)
          local selectedRepo = repoNames[position + 1]
          if repoDialog then repoDialog.dismiss() end
          showRepoContent(selectedRepo, "")
        end
        
        itemsIds.repoList.onItemLongClick = function(parent, view, position, id)
          local selectedRepo = repoNames[position + 1]
          if repoDialog then repoDialog.dismiss() end
          showRepoOptions(selectedRepo, function() showRepoList() end)
          return true
        end
      end
    end
    
    local itemsLayout = {
      LinearLayout, orientation = "vertical", layout_width = "fill", layout_height = "fill", padding = "10dp",
      {
        LinearLayout, orientation = "horizontal", layout_width = "fill", layout_marginBottom = "10dp", gravity = "center_vertical",
        { TextView, id = "txtRepoCount", text = "My All Repositories", textSize = "18sp", layout_weight = "1", typeface = Typeface.DEFAULT_BOLD },
        { Button, text = "Search", layout_marginRight = "5dp", onClick = function()
            local searchIds = {}
            local searchBuilder = AlertDialog.Builder(ctx)
            searchBuilder.setTitle("Search Repositories")
            searchBuilder.setView(loadlayout({
              LinearLayout, orientation = "vertical", padding = "20dp",
              { EditText, id = "searchInput", hint = "Enter repository name...", text = currentSearchQuery, layout_width = "fill" }
            }, searchIds))
            searchBuilder.setNeutralButton("Search", function()
              currentSearchQuery = searchIds.searchInput.getText().toString()
              applyFiltersAndSort()
            end)
            searchBuilder.setNegativeButton("Clear", function()
              currentSearchQuery = ""
              applyFiltersAndSort()
            end)
            searchBuilder.setPositiveButton("Cancel", nil)
            searchBuilder.show()
          end 
        },
        { Button, text = "Create", backgroundColor = 0xFF28A745, textColor = 0xFFFFFFFF, onClick = function()

            filterBuilder.setItems(typeOptions, function(dialog, which)

        { Button, id = "btnSortFilter", text = "Sort: Recently pushed", layout_weight = "1", layout_marginLeft = "2dp", onClick = function()

      if keyCode == KeyEvent.KEYCODE_BACK and event.getAction() == KeyEvent.ACTION_UP then
        if repoDialog then repoDialog.dismiss() end
        if activity then activity.finish() end
        return true
      end
      return false
    end)
    
    repoDialog.show()
    applyFiltersAndSort()
  end
  
  if preloadedData.repos then
      handleResponse(200, preloadedData.repos, nil)
      return
  end

            if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
            speak("No internet connection. Please check your internet connection and try again later.")
            return
        end
        handleResponse(code, body, loadingDlg)
      end
    })

function executeDeleteFolder(repoName, folderPath)
    local userName = ctx.getSharedData("git_user")
    local token = ctx.getSharedData("git_token")

    local loadingDlg = LuaDialog(ctx)
    loadingDlg.setCancelable(false)
    loadingDlg.setView(loadlayout({
        LinearLayout, orientation = "vertical", padding = "30dp", gravity = "center",
        { ProgressBar, layout_gravity = "center" },
        { TextView, text = "Deleting folder...", layout_gravity = "center", layout_marginTop = "10dp" }
    }))
    loadingDlg.show()

    local headers = {
        ["Authorization"] = "Bearer " .. token,
        ["Accept"] = "application/vnd.github.v3+json",
        ["User-Agent"] = "GitHub-Manager-App"
    }
    
    local safePath = Uri.encode(folderPath, "/")
    local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. safePath

    Http.get(url, headers, function(code, body)
        if code == 200 then
            local success, files = pcall(function() return json.decode(body) end)
            if success and type(files) == "table" then
                local filesToDelete = {}
                for i = 1, #files do
                    if files[i].type == "file" then
                        table.insert(filesToDelete, files[i])
                    end
                end

                if #filesToDelete == 0 then
                    Handler(Looper.getMainLooper()).post(Runnable{
                        run = function()
                            if loadingDlg then loadingDlg.dismiss() end
                            showToastMessage("Cannot delete empty or complex nested folder automatically.")
                        end
                    })
                    return
                end

                local deletedCount = 0
                local hasError = false

                local function deleteNextFile()
                    if deletedCount >= #filesToDelete then
                        Handler(Looper.getMainLooper()).post(Runnable{
                            run = function()
                                if loadingDlg then loadingDlg.dismiss() end
                                if not hasError then
                                    local parentPath = string.match(folderPath, "^(.*)/[^/]+$") or ""
                                    preloadedData.repoContents[tostring(repoName) .. "|" .. parentPath] = nil
                                    preloadedData.repoContents[tostring(repoName) .. "|" .. folderPath] = nil
                                    preloadedData.repoContents[tostring(repoName) .. "|"] = nil

                                    local sBuilder = AlertDialog.Builder(ctx)
                                    sBuilder.setTitle("Success")
                                    sBuilder.setMessage("Folder deleted successfully!")
                                    sBuilder.setPositiveButton("OK", function()
                                        if contentDialog then contentDialog.dismiss() end
                                        showRepoContent(repoName, parentPath)
                                    end)
                                    sBuilder.show()
                                end
                            end
                        })
                        return
                    end

                    local f = filesToDelete[deletedCount + 1]
                    local delBody = json.encode({ message = "Delete " .. f.name .. " via GitHub by Tech For VI", sha = f.sha })
                    local fUrl = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. Uri.encode(f.path, "/")

                    customHttpRequest(fUrl, "DELETE", headers, delBody, function(dCode, dBody)
                        if dCode ~= 200 and dCode ~= 204 then
                            hasError = true
                            Handler(Looper.getMainLooper()).post(Runnable{
                                run = function()
                                    if loadingDlg then loadingDlg.dismiss() end
                                    local errBuilder = AlertDialog.Builder(ctx)
                                    errBuilder.setTitle("Error")
                                    errBuilder.setMessage("Failed to delete file inside folder.\n" .. getErrorMessage(dCode, dBody))
                                    errBuilder.setPositiveButton("OK", nil)
                                    errBuilder.show()
                                end
                            })
                            return
                        end
                        deletedCount = deletedCount + 1
                        deleteNextFile()
                    end)
                end

                deleteNextFile()
            else
                Handler(Looper.getMainLooper()).post(Runnable{
                    run = function()
                        if loadingDlg then loadingDlg.dismiss() end
                        showToastMessage("Failed to parse folder contents.")
                    end
                })
            end
        else
            Handler(Looper.getMainLooper()).post(Runnable{
                run = function()
                    if loadingDlg then loadingDlg.dismiss() end
                    showToastMessage("Error accessing folder.")
                end
            })
        end
    end)
end

function commitNewFile(repoName, path, fileName, content, b64Content)
  local userName = ctx.getSharedData("git_user")
  local token = ctx.getSharedData("git_token")
  
  local loadingDlg = LuaDialog(ctx)
  loadingDlg.setCancelable(false)
  loadingDlg.setView(loadlayout({
    LinearLayout, orientation = "vertical", padding = "30dp", gravity = "center",
    { ProgressBar, layout_gravity = "center" },
    { TextView, text = "Committing file...", layout_gravity = "center", layout_marginTop = "10dp" }
  }))
  loadingDlg.show()
  
  local headers = {
    ["Authorization"] = "Bearer " .. token,
    ["Accept"] = "application/vnd.github.v3+json",
    ["Content-Type"] = "application/json",
    ["User-Agent"] = "GitHub-Manager-App"
  }
  
  local encoded = b64Content or encodeBase64(content)
  local body = json.encode({ message = "Create " .. fileName .. " via GitHub by Tech For VI", content = encoded })
  
  local fullPath = (path == nil or path == "") and fileName or (path .. "/" .. fileName)
  fullPath = fullPath:gsub("//+", "/")
  
  local safePath = Uri.encode(fullPath, "/")
  local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. safePath
  
  customHttpRequest(url, "PUT", headers, body, function(code, respBody)
    Handler(Looper.getMainLooper()).post(Runnable{
      run = function()
        if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
        if code == -1 or code == 0 then
            speak("No internet connection. Please check your internet connection and try again later.")
            return
        end
        if code == 200 or code == 201 then
          
          local parentPath = string.match(path, "^(.*)/[^/]+$") or ""
          if path == "" then parentPath = "" end
          preloadedData.repoContents[tostring(repoName) .. "|" .. tostring(path)] = nil
          preloadedData.repoContents[tostring(repoName) .. "|" .. parentPath] = nil
          preloadedData.repoContents[tostring(repoName) .. "|"] = nil
          
          local sBuilder = AlertDialog.Builder(ctx)
          sBuilder.setTitle("Success")
          sBuilder.setMessage("File committed successfully!")
          sBuilder.setPositiveButton("OK", function()
            if contentDialog then contentDialog.dismiss() end
            showRepoContent(repoName, path)
          end)
          sBuilder.show()
        else
          local errBuilder = AlertDialog.Builder(ctx)
          errBuilder.setTitle("Error")
          errBuilder.setMessage(getErrorMessage(code, respBody))
          errBuilder.setPositiveButton("OK", nil)
          errBuilder.show()
        end
      end
    })
  end)
end

function processSelectedFile(uri)
  if not globalUploadRepo then return end
  local fileName = "uploaded_file"
  local cursor = ctx.getContentResolver().query(uri, nil, nil, nil, nil)
  if cursor and cursor.moveToFirst() then
    local nameIndex = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
    if nameIndex ~= -1 then
      fileName = cursor.getString(nameIndex)
    end
    cursor.close()
  end
  
  local success, b64Content = pcall(function()
    local is = ctx.getContentResolver().openInputStream(uri)
    local bos = ByteArrayOutputStream()
    local b = byte[4096]
    local len = is.read(b)
    while len ~= -1 do
      bos.write(b, 0, len)
      len = is.read(b)
    end
    local fileBytes = bos.toByteArray()
    local encodedStr = Base64.encodeToString(fileBytes, Base64.NO_WRAP)
    is.close()
    bos.close()
    return encodedStr
  end)
  
  if not success then
    local errBuilder = AlertDialog.Builder(ctx)
    errBuilder.setTitle("Error")
    errBuilder.setMessage("Could not read file for upload.")
    errBuilder.setPositiveButton("OK", nil)
    errBuilder.show()
    return
  end
  
  local confirmBuilder = AlertDialog.Builder(ctx)
  confirmBuilder.setTitle("Upload File")
  confirmBuilder.setMessage("File chosen: " .. fileName .. "\n\nDo you want to commit this file?")
  
  confirmBuilder.setNegativeButton("Commit", function()
    commitNewFile(globalUploadRepo, globalUploadPath, fileName, nil, b64Content)
  end)
  confirmBuilder.setPositiveButton("Cancel", nil)
  confirmBuilder.show()
end

function onActivityResult(requestCode, resultCode, data)
  if requestCode == 2002 and resultCode == -1 and data then
    local uri = data.getData()
    processSelectedFile(uri)
  end
end

function showRepoContent(repoName, path)
  local token = ctx.getSharedData("git_token")
  local userName = ctx.getSharedData("git_user")
  path = path or ""
  
  local function handleResponse(code, body, loadingDlg)
    if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
    
    local files = {}
    if code == 404 then
    elseif code ~= 200 then
      local errBuilder = AlertDialog.Builder(ctx)
      errBuilder.setTitle("Error")
      errBuilder.setMessage(getErrorMessage(code, body))
      errBuilder.setPositiveButton("OK", function() showRepoList() end)
      errBuilder.show()
      return
    else
      local success, parsed = pcall(function() return json.decode(body) end)
      if success and type(parsed) == "table" then
        files = parsed
      end
    end
    
    local fileData = {}
    local fileObjects = {}
    if #files > 0 then
      for i = 1, #files do
        local f = files[i]
        if f and f.name then
          local fType = f.type == "dir" and "Folder" or "File"
          table.insert(fileData, { fileNameTxt = tostring(f.name), fileTypeTxt = fType })
          table.insert(fileObjects, f)
        end
      end
    end
    
    local itemsIds = {}
    local itemsLayout = {
      LinearLayout, orientation = "vertical", layout_width = "fill", layout_height = "fill", padding = "10dp",
      {
        FrameLayout, layout_width = "fill", layout_height = "0dp", layout_weight = "1",
        { ListView, id = "fileList", layout_width = "fill", layout_height = "fill", fastScrollEnabled = true, dividerHeight = "1px" },
        { TextView, id = "emptyMsg", text = "This folder is empty.", textSize = "16sp", textColor = 0xFF888888, layout_gravity = "center", visibility = View.GONE }
      },
      {
        LinearLayout, orientation = "horizontal", layout_width = "fill", layout_marginTop = "10dp",
        { Button, text = "CREATE", layout_weight = "1", layout_marginRight = "2dp", onClick = function() 
            local optBuilder = AlertDialog.Builder(ctx)
            optBuilder.setTitle("Create")
            optBuilder.setItems({"Create File", "Create Folder"}, function(dialog, which)
              if which == 0 then
                startCreateFile(repoName, path)
              elseif which == 1 then
                startCreateFolder(repoName, path)
              end
            end)
            optBuilder.show()
          end 
        },
        { Button, text = "UPLOAD FILE", layout_weight = "1", layout_marginLeft = "2dp", layout_marginRight = "2dp", onClick = function() startUploadFile(repoName, path) end },
        { Button, text = "SHARE", layout_weight = "1", layout_marginLeft = "2dp", layout_marginRight = "2dp", onClick = function() shareRepository(repoName) end },
        { Button, text = "GO BACK", layout_weight = "1", layout_marginLeft = "2dp", onClick = function()
            if contentDialog then contentDialog.dismiss() end
            if path == "" then showRepoList()
            else showRepoContent(repoName, string.match(path, "^(.*)/[^/]+$") or "") end
          end 
        }
      }
    }
    
    contentDialog = LuaDialog(ctx)
    contentDialog.setView(loadlayout(itemsLayout, itemsIds))
    contentDialog.setTitle(tostring(repoName))
    
    contentDialog.setOnKeyListener(function(dialog, keyCode, event)
      if keyCode == KeyEvent.KEYCODE_BACK and event.getAction() == KeyEvent.ACTION_UP then
        if contentDialog then contentDialog.dismiss() end
        if path == "" then showRepoList()
        else showRepoContent(repoName, string.match(path, "^(.*)/[^/]+$") or "") end
        return true
      end
      return false
    end)
    
    contentDialog.show()
    
    if #fileData == 0 then
      itemsIds.fileList.setVisibility(View.GONE)
      itemsIds.emptyMsg.setVisibility(View.VISIBLE)
    else
      local adapterLayout = {
        LinearLayout, orientation = "vertical", layout_width = "fill", padding = "15dp",
        { TextView, id = "fileNameTxt", textSize = "16sp", textColor = 0xFF000000, typeface = Typeface.DEFAULT_BOLD },
        { TextView, id = "fileTypeTxt", textSize = "12sp", textColor = 0xFF666666, layout_marginTop = "2dp" }
      }
      local adapter = LuaAdapter(ctx, fileData, adapterLayout)
      itemsIds.fileList.setAdapter(adapter)
      
      itemsIds.fileList.onItemClick = function(parent, view, position, id)
        local item = fileObjects[position + 1]
        if item.type == "dir" then
          if contentDialog then contentDialog.dismiss() end
          showRepoContent(repoName, item.path)
        elseif item.type == "file" then openFileContent(repoName, item) end
      end
      
      itemsIds.fileList.onItemLongClick = function(parent, view, position, id)
        local item = fileObjects[position + 1]
        if item.type == "file" then
          local optBuilder = AlertDialog.Builder(ctx)
          optBuilder.setTitle(tostring(item.name))
          optBuilder.setItems({"Edit File", "Delete File", "Share File", "Cancel"}, function(dialog, which)
            if which == 0 then openFileContent(repoName, item)
            elseif which == 1 then deleteFileConfirm(repoName, item)
            elseif which == 2 then
              local shareIntent = Intent(Intent.ACTION_SEND)
              shareIntent.setType("text/plain")
              shareIntent.putExtra(Intent.EXTRA_TEXT, item.html_url)
              ctx.startActivity(Intent.createChooser(shareIntent, "Share File Link"))
            end
          end)
          optBuilder.show()
        elseif item.type == "dir" then
          local optBuilder = AlertDialog.Builder(ctx)
          optBuilder.setTitle(tostring(item.name))
          optBuilder.setItems({"Open", "Delete", "Cancel"}, function(dialog, which)
            if which == 0 then
              if contentDialog then contentDialog.dismiss() end
              showRepoContent(repoName, item.path)
            elseif which == 1 then
              deleteFolderConfirm(repoName, item)
            end
          end)
          optBuilder.show()
        end
        return true
      end
    end
  end

  local cacheKey = tostring(repoName) .. "|" .. tostring(path)
  if preloadedData.repoContents[cacheKey] then
      handleResponse(200, preloadedData.repoContents[cacheKey], nil)
      return
  end
  
  local loadingDlg = LuaDialog(ctx)
  loadingDlg.setCancelable(false)
  loadingDlg.setView(loadlayout({
    LinearLayout, orientation = "vertical", padding = "30dp", layout_width = "fill", gravity = "center",
    { ProgressBar, layout_gravity = "center" },
    { TextView, text = "Loading files...", layout_gravity = "center", layout_marginTop = "10dp" }
  }))
  loadingDlg.show()
  
  local headers = {
    ["Authorization"] = "Bearer " .. token,
    ["Accept"] = "application/vnd.github.v3+json",
    ["User-Agent"] = "GitHub-Manager-App"
  }
  
  local safePath = Uri.encode(path, "/")
  local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. safePath
  
  Http.get(url, headers, function(code, body)
    Handler(Looper.getMainLooper()).post(Runnable{
      run = function()
        if code == -1 or code == 0 then
            if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
            speak("No internet connection. Please check your internet connection and try again later.")
            return
        end
        handleResponse(code, body, loadingDlg)
      end
    })
  end)
end

function openFileContent(repoName, fileItem)
  local token = ctx.getSharedData("git_token")
  
  local function handleResponse(code, body, loadingDlg)
    if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
    if code ~= 200 then
      local errBuilder = AlertDialog.Builder(ctx)
      errBuilder.setTitle("Error")
      errBuilder.setMessage(getErrorMessage(code, body))
      errBuilder.setPositiveButton("OK", nil)
      errBuilder.show()
      return
    end
    
    local fileIds = {}
    local fileLayout = {
      LinearLayout, orientation = "vertical", layout_width = "fill", layout_height = "fill", padding = "10dp",
      { TextView, text = tostring(fileItem.name), textSize = "18sp", layout_gravity = "center", layout_marginBottom = "10dp", typeface = Typeface.DEFAULT_BOLD },
      {
        ScrollView,
        layout_width = "fill",
        layout_height = "0dp",
        layout_weight = 1,
        {
          EditText,
          id = "fileContentInput",
          layout_width = "fill",
          layout_height = "wrap_content",
          text = tostring(body),
          gravity = "top",
          textSize = "14sp",
          lines = 1,
          horizontallyScrolling = true,
        }
      },
      { EditText, id = "commitMessageInput", hint = "Commit message...", layout_width = "fill", layout_marginTop = "5dp", layout_marginBottom = "5dp" },
      {
        LinearLayout, orientation = "horizontal", layout_width = "fill", layout_marginTop = "5dp",
        { Button, text = "Commit Changes", layout_weight = "1", layout_marginRight = "5dp", backgroundColor = 0xFF28A745, textColor = 0xFFFFFFFF, onClick = function()
            updateFileContent(repoName, fileItem, fileIds.fileContentInput.getText().toString(), fileIds.commitMessageInput.getText().toString())
          end },
        { Button, text = "Go Back", layout_weight = "1", layout_marginLeft = "5dp", onClick = function()
            if fileDialog then fileDialog.dismiss() end
          end }
      }
    }
    
    fileDialog = LuaDialog(ctx)
    fileDialog.setView(loadlayout(fileLayout, fileIds))
    
    fileDialog.setOnKeyListener(function(dialog, keyCode, event)
      if keyCode == KeyEvent.KEYCODE_BACK and event.getAction() == KeyEvent.ACTION_UP then
        if fileDialog then fileDialog.dismiss() end
        return true
      end
      return false
    end)
    
    fileDialog.show()
  end

  if fileItem.sha and preloadedData.fileContents[fileItem.sha] then
      handleResponse(200, preloadedData.fileContents[fileItem.sha], nil)
      return
  end
  
  local loadingDlg = LuaDialog(ctx)
  loadingDlg.setCancelable(false)
  loadingDlg.setView(loadlayout({
    LinearLayout, orientation = "vertical", padding = "30dp", gravity = "center",
    { ProgressBar, layout_gravity = "center" },
    { TextView, text = "Loading content...", layout_gravity = "center", layout_marginTop = "10dp" }
  }))
  loadingDlg.show()
  
  local headers = {
    ["Authorization"] = "Bearer " .. token,
    ["Accept"] = "application/vnd.github.v3.raw",
    ["User-Agent"] = "GitHub-Manager-App"
  }
  
  Http.get(fileItem.download_url, headers, function(code, body)
    Handler(Looper.getMainLooper()).post(Runnable{
      run = function()
        if code == -1 or code == 0 then
            if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
            speak("No internet connection. Please check your internet connection and try again later.")
            return
        end
        handleResponse(code, body, loadingDlg)
      end
    })
  end)
end

function updateFileContent(repoName, fileItem, newContent, commitMessage)
  local userName = ctx.getSharedData("git_user")
  local token = ctx.getSharedData("git_token")
  if commitMessage == "" then commitMessage = "Update " .. tostring(fileItem.name) .. " via GitHub by Tech For VI" end
  
  local loadingDlg = LuaDialog(ctx)
  loadingDlg.setView(loadlayout({
    LinearLayout, orientation = "vertical", padding = "30dp", gravity = "center",
    { ProgressBar, layout_gravity = "center" },
    { TextView, text = "Committing changes...", layout_gravity = "center", layout_marginTop = "10dp" }
  }))
  loadingDlg.show()
  
  local headers = {
    ["Authorization"] = "Bearer " .. token,
    ["Accept"] = "application/vnd.github.v3+json",
    ["Content-Type"] = "application/json",
    ["User-Agent"] = "GitHub-Manager-App"
  }
  
  local body = json.encode({ message = commitMessage, content = encodeBase64(newContent), sha = fileItem.sha })
  local safePath = Uri.encode(fileItem.path, "/")
  local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. safePath
  
  customHttpRequest(url, "PUT", headers, body, function(code, respBody)
    Handler(Looper.getMainLooper()).post(Runnable{
      run = function()
        if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
        if code == -1 or code == 0 then
            speak("No internet connection. Please check your internet connection and try again later.")
            return
        end
        if code == 200 or code == 201 then
          
          local parentPath = string.match(fileItem.path, "^(.*)/[^/]+$") or ""
          preloadedData.repoContents[tostring(repoName) .. "|" .. parentPath] = nil
          preloadedData.fileContents[fileItem.sha] = nil
          
          local sBuilder = AlertDialog.Builder(ctx)
          sBuilder.setTitle("Success")
          sBuilder.setMessage("Changes committed successfully!")
          sBuilder.setPositiveButton("OK", function()
              if fileDialog then fileDialog.dismiss() end
              if contentDialog then contentDialog.dismiss() end
              showRepoContent(repoName, parentPath)
          end)
          sBuilder.show()
        else
          local errBuilder = AlertDialog.Builder(ctx)
          errBuilder.setTitle("Error")
          errBuilder.setMessage(getErrorMessage(code, respBody))
          errBuilder.setPositiveButton("OK", nil)
          errBuilder.show()
        end
      end
    })
  end)
end

function deleteFileConfirm(repoName, fileItem)
  local confirmBuilder = AlertDialog.Builder(ctx)
  confirmBuilder.setTitle("Delete File")
  confirmBuilder.setMessage("Are you sure you want to delete '" .. tostring(fileItem.name) .. "'?")
  
  confirmBuilder.setNegativeButton("Delete", function()
    local userName = ctx.getSharedData("git_user")
    local token = ctx.getSharedData("git_token")
    
    local loadingDlg = LuaDialog(ctx)
    loadingDlg.setView(loadlayout({
      LinearLayout, orientation = "vertical", padding = "30dp", gravity = "center",
      { ProgressBar, layout_gravity = "center" },
      { TextView, text = "Deleting...", layout_gravity = "center", layout_marginTop = "10dp" }
    }))
    loadingDlg.show()
    
    local safePath = Uri.encode(fileItem.path, "/")
    local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName) .. "/contents/" .. safePath
    local headers = {
      ["Authorization"] = "Bearer " .. token,
      ["Accept"] = "application/vnd.github.v3+json",
      ["User-Agent"] = "GitHub-Manager-App"
    }
    
    Http.get(url, headers, function(code, body)
      Handler(Looper.getMainLooper()).post(Runnable{
        run = function()
          if code == -1 or code == 0 then
              if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
              speak("No internet connection. Please check your internet connection and try again later.")
              return
          end
          if code == 200 then
            local success, data = pcall(function() return json.decode(body) end)
            if success and data and data.sha then
              local latestSha = data.sha
              
              local delHeaders = {
                ["Authorization"] = "Bearer " .. token,
                ["Accept"] = "application/vnd.github.v3+json",
                ["Content-Type"] = "application/json",
                ["User-Agent"] = "GitHub-Manager-App"
              }
              local delBody = json.encode({ message = "Delete " .. fileItem.name .. " via GitHub by Tech For VI", sha = latestSha })
              
              customHttpRequest(url, "DELETE", delHeaders, delBody, function(dCode, dBody)
                Handler(Looper.getMainLooper()).post(Runnable{
                  run = function()
                    if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
                    if dCode == -1 or dCode == 0 then
                        speak("No internet connection. Please check your internet connection and try again later.")
                        return
                    end
                    if dCode == 200 or dCode == 204 then
                      
                      local parentPath = string.match(fileItem.path, "^(.*)/[^/]+$") or ""
                      preloadedData.repoContents[tostring(repoName) .. "|" .. parentPath] = nil
                      
                      local sBuilder = AlertDialog.Builder(ctx)
                      sBuilder.setTitle("Success")
                      sBuilder.setMessage("File deleted successfully!")
                      sBuilder.setPositiveButton("OK", function()
                        if fileDialog then fileDialog.dismiss() end
                        if contentDialog then contentDialog.dismiss() end
                        showRepoContent(repoName, string.match(fileItem.path, "^(.*)/[^/]+$") or "")
                      end)
                      sBuilder.show()
                    else
                      local errBuilder = AlertDialog.Builder(ctx)
                      errBuilder.setTitle("Error")
                      errBuilder.setMessage("Failed to delete file.\n" .. getErrorMessage(dCode, dBody))
                      errBuilder.setPositiveButton("OK", nil)
                      errBuilder.show()
                    end
                  end
                })
              end)
            else
              if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
              local errBuilder = AlertDialog.Builder(ctx)
              errBuilder.setTitle("Error")
              errBuilder.setMessage("Could not parse file data for deletion.")
              errBuilder.setPositiveButton("OK", nil)
              errBuilder.show()
            end
          else
            if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
            local errBuilder = AlertDialog.Builder(ctx)
            errBuilder.setTitle("Error")
            errBuilder.setMessage("Could not fetch latest file status. Error: " .. tostring(code) .. "\n" .. getErrorMessage(code, body))
            errBuilder.setPositiveButton("OK", nil)
            errBuilder.show()
          end
        end
      })
    end)
  end)
  confirmBuilder.setPositiveButton("Cancel", nil)
  confirmBuilder.show()
end

    loadingDlg.show()
    
    local headers = { ["Authorization"] = "Bearer " .. token, ["Accept"] = "application/vnd.github.v3+json", ["Content-Type"] = "application/json", ["User-Agent"] = "GitHub-Manager-App" }
    local url = "https://api.github.com/repos/" .. userName .. "/" .. tostring(repoName)
    
    customHttpRequest(url, "PATCH", headers, json.encode({ private = setPrivate }), function(code, respBody)
      Handler(Looper.getMainLooper()).post(Runnable{
        run = function()
          if loadingDlg then pcall(function() loadingDlg.dismiss() end) end
          if code == -1 or code == 0 then
              speak("No internet connection. Please check your internet connection and try again later.")
              return
          end
          if code == 200 then
            local sBuilder = AlertDialog.Builder(ctx)
            sBuilder.setTitle("Success")
            sBuilder.setMessage("Visibility changed")
            sBuilder.setPositiveButton("OK", function() if onComplete then onComplete() end end)
            sBuilder.show()
          else
            local errBuilder = AlertDialog.Builder(ctx)
            errBuilder.setTitle("Error")
            errBuilder.setMessage(getErrorMessage(code, respBody))
            errBuilder.setPositiveButton("OK", function() if onComplete then onComplete() end end)
            errBuilder.show()
          end
        end
      })
    end)
  end)
  visBuilder.setPositiveButton("Cancel", function() if onComplete then onComplete() end end)
  
  visBuilder.setOnCancelListener(function(dialog)
    if onComplete then onComplete() end
  end)
  
  visBuilder.show()
end

  startInitialPreload(savedToken, savedUser)
